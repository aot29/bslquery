 20    def __init__(self, db_config, verbose=False):
 21        self.db_config = db_config
 22        self.verbose = verbose  # If True, print messages to stdout (for Jupyter)
 23        # logger tied to the concrete subclass's module (Pythonesque way),
 24        # so each subclass logs under its own name, e.g. bslquery.BioSoundsLabDB
 25        self._logger = logging.getLogger(type(self).__module__)

 28    def _log(self, message, level=logging.INFO):
 29        """Log a message, and print if verbose is True."""
 30        if self.verbose:
 31            print(message)
 32        if level == logging.ERROR:
 33            self._logger.error(message)
 34        else:
 35            self._logger.info(message)

 38    def _connect(self):
 39        """Open a new connection using self.db_config.
 40
 41        Single seam patched in unit tests. Shared by all subclasses;
 42        override in a subclass only to change how connections are made.
 43        """
 44        return psycopg2.connect(**self.db_config)

 48        """
 49        Test if a connection to the PostgreSQL database can be established.
 50        db_config (dict): Database connection parameters (host, port, user, password, database).
 51        returns: True if the connection is successful, False otherwise.
 52        """
 53        try:
 54            with self._connect() as conn:
 55                self._log("Connection successful!")
 56                return True
 57        except OperationalError as e:
 58            self._log(f"Connection failed: {e}")
 59            return False
 60        except DatabaseError as e:
 61            self._log(f"Database error: {e}")
 62            return False

 72    @staticmethod
 73    def _rows_as_dicts(cursor):
 74        """fetchall() as a list of dicts keyed by column name, or None if no rows."""
 75        rows = cursor.fetchall()
 76        if not rows:
 77            return None                      # tests assert empty result -> None
 78        columns = [d[0] for d in cursor.description]
 79        return [dict(zip(columns, row)) for row in rows]

 25    @staticmethod
 26    def days_between_dates(datestr1, datestr2):
 27        """
 28        Helper function to count days between dates.
 29        The start date and the end date are included in the count.
 30        This method does not query the database. To get the number of dates with a recording, use days_with_recordings instead.
 31        Static: callable as BioSoundsLabDB.days_between_dates(...) without an instance.
 32
 33        date1, date2: string in the form YYYY-MM-DD
 34        return int number of days between dates
 35        """
 36        date1 = datetime.strptime(datestr1, '%Y-%m-%d')
 37        date2 = datetime.strptime(datestr2, '%Y-%m-%d')
 38        # Calculate the difference
 39        delta = abs((date2 - date1).days) + 1
 40
 41        # Get the absolute number of days
 42        return delta

 63    def fetch_combined_counts_breeding_season(self, species_name, site_prefix, start_date_breeding_season, end_date_breeding_season):
 64        """
 65        Combined days in breeding season and days with recordings.
 66        The purpose of this query is to compare recordings counts within a period of time, mostly the breeding season.
 67
 68        Parameters:
 69        * species_name string: scientific name ScientificName_IOC10.1
 70        * site_prefix string: site prefix, e.g. BBBRI01
 71        * start_date_breeding_season string: date string (YYYY-MM-DD),
 72        * end_date_breeding_season: date string (YYYY-MM-DD)
 73
 74        returns a dict in the form:
 75        ScientificName_IOC10.1, a string
 76        StartDate breeding season (MM-DD),
 77        EndDate breeding season (MM-DD),
 78        Days in breeding season (one column for each year),
 79        Site prefix year recording count (one column for each combination of prefix and year)
 80        """
 81        values = {
 82            'species_name': species_name,
 83            'site_prefix': site_prefix,
 84            'start_date_breeding_season': start_date_breeding_season,
 85            'end_date_breeding_season': end_date_breeding_season,
 86        }
 87        site_id = self.fetch_site_id_by_prefix(site_prefix)
 88        values['days_in_breeding_season'] = self.days_between_dates(
 89            values['start_date_breeding_season'], values['end_date_breeding_season']
 90        )
 91        values['days_with_recordings'] = self.days_with_recordings(
 92            values['start_date_breeding_season'], values['end_date_breeding_season'], site_id
 93        )
 94        return values

18    def _row_to_dwc(row):
19        """Map one inference row (dict of column name -> value) to a Darwin Core
20        Occurrence record: DwC term names as keys, non-DwC extras (confidence,
21        thresholds, segment times) packed into dynamicProperties as JSON."""
22        return {
23            "occurrenceID": f"{row['filename']}#{row['start_time']}",
24            "eventID": row["filename"],
25            "eventDate": str(row["date"]),
26            "eventTime": str(row["time"]),
27            "scientificName": row["species"],
28            "taxonRank": "species",
29            "occurrenceStatus": "present",
30            "basisOfRecord": "MachineObservation",
31            "recordedBy": row["model_name"],
32            "locationID": row["site_prefix"],
33            "locality": row["site_name"],
34            "dynamicProperties": json.dumps({
35                "start_time": str(row["start_time"]),
36                "end_time": str(row["end_time"]),
37                "confidence": row["confidence"],
38                "experimental_threshold": row["experimental_threshold"],
39                "preliminary_threshold": row["preliminary_threshold"],
40                "final_threshold": row["final_threshold"],
41            }),
42        }

46    def fetch_inferences(self, cursor, species_name, model_name, site_prefix, year, min_confidence=0):
47        """
48        Fetch all inferences for this combination of species, site and year.

50        row (see _row_to_dwc), or None if there are no results.

52        cursor.execute(queries.FETCH_SITE_ID_BY_PREFIX, (f"%{site_prefix}%",))

55        label_id = self._fetchone_scalar(cursor)

72        results = [self._row_to_dwc(dict(zip(columns, row))) for row in rows]

74        return results

13    """
14    Validates Darwin Core (DwC) data against the BioSoundsLab occurrence PAM
15    schema (schema/dwc-occurrence-pam.schema.json).
16
17    Pure JSON-schema checking: no database connection is needed, so this
18    deliberately does NOT extend BioSoundsLabBase.
19    """
20    # repo-root schema/, resolved from this file so it works from any CWD
21    SCHEMA_FILE = Path(__file__).resolve().parents[2] / "schema" / "dwc-occurrence-pam.schema.json"
22
23    def __init__(self, verbose=False):
24        self.verbose = verbose  # If True, print messages to stdout (for Jupyter)
25
26    def _log(self, message, level=logging.INFO):
27        """Log a message, and print if verbose is True."""
28        if self.verbose:
29            print(message)
30        if level == logging.ERROR:
31            logger.error(message)
32        else:
33            logger.info(message)
34
35    def validate(self, data):
36        """
37        Validate Darwin Core (DwC) data against the BioSoundsLab occurrence PAM
38        schema. `data` is a list of occurrence records, e.g. the result of
39        BioSoundsLabDwC.fetch_inferences.
40        Returns: True if the data is valid, False otherwise (errors are logged).
41        """
42        with open(self.SCHEMA_FILE) as f:
43            schema = json.load(f)
44        validator = jsonschema.Draft7Validator(schema)
45        errors = sorted(validator.iter_errors(data), key=lambda e: list(e.absolute_path))
46        for e in errors:
47            self._log(f"DwC validation error at {list(e.absolute_path)}: {e.message}", level=logging.ERROR)
48        return not errors
